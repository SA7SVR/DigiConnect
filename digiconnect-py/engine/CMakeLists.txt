cmake_minimum_required(VERSION 3.16)
project(dcengine VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
find_package(Qt6 REQUIRED COMPONENTS Core Multimedia Network SerialPort)
file(GLOB_RECURSE SRC CONFIGURE_DEPENDS src/*.cpp src/*.cc)
add_executable(dcengine MACOSX_BUNDLE dcengine.cpp dcengine.h ${SRC})
set_target_properties(dcengine PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.hellaszone.dcengine MACOSX_BUNDLE_BUNDLE_NAME dcengine
  MACOSX_BUNDLE_INFO_STRING "DigiConnect protocol engine" MACOSX_BUNDLE_BUNDLE_VERSION 1.0 MACOSX_BUNDLE_SHORT_VERSION_STRING 1.0
  MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in)
target_include_directories(dcengine PRIVATE src)
target_compile_definitions(dcengine PRIVATE VERSION_NUMBER="1")
target_link_libraries(dcengine PRIVATE Qt6::Core Qt6::Multimedia Qt6::Network Qt6::SerialPort)
if(WIN32)
  target_link_libraries(dcengine PRIVATE ws2_32)
endif()
if(UNIX)
  target_link_libraries(dcengine PRIVATE dl)
endif()
